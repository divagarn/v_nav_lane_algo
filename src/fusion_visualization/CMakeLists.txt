cmake_minimum_required(VERSION 3.8)
project(fusion_visualization)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(perception_msgs REQUIRED)
find_package(fusion_common REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc highgui)
fusion_opencv_libs(OpenCV_LIBS opencv_core opencv_imgproc opencv_highgui)   # from fusion_common (handles configs without variables)

add_executable(fusion_visualization_node src/fusion_visualization_node.cpp)
target_link_libraries(fusion_visualization_node fusion_common::fusion_common ${OpenCV_LIBS})
ament_target_dependencies(fusion_visualization_node rclcpp sensor_msgs visualization_msgs geometry_msgs perception_msgs)

install(TARGETS fusion_visualization_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
